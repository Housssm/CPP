#ifndef ITER_HPP
#define ITER_HPP
#include <iostream>
#include <cstddef>

template <typename T, typename F> 
void iter(T *array, const size_t arrayLen, F f)
{
	if (!array || !f)
		return;
	for ( size_t i = 0; i < arrayLen; i++)
		f(array[i]);
}

#endif

#include "Iter.hpp"


void	add(int &i)
{
	i += 1;
}

void	suffix(std::string &value)
{
	value = value + "ix";
}

template<typename T>
void	Print(const T &value)
{
	std::cout << value << std::endl;
}

int main ()
{
	std::cout << "\n---TEST SIMPLE ---" << std::endl;

	int num[4] = {1,3,5,7};
	std::string word[3] = {"Idef", "Obel", "Panoram"};
	
	iter(num, 4, add);
	iter(word, 3, suffix);
	iter(num, 4, Print<int>);
	iter(word, 3, Print<std::string>);

	std::cout << "\n---TEST TABLEAU CONST/VIDE ---" << std::endl;

	const int num2[4] = {10,30,40,50};
	int	empty_arr[1] = {0};

	iter(empty_arr, 0, add);
	iter(num2, 4, Print<int>);
	iter(empty_arr, 0, Print<int>);
	
	
	return 0;
}

#ifndef ARRAY_HPP
#define ARRAY_HPP
#include <iostream>
#include <exception>
#include <cstdlib>

template <typename T>
class Array
{
	public :
		Array() : _elements(NULL), _size(0)
		{
		}

		Array(unsigned int n) : _elements(new T[n]()), _size(n)
		{
		}

		Array(const Array &other) : _elements(NULL), _size(0)
		{
			if (other._size == 0)
			{
				_size = 0;
				_elements = NULL;
			}
			else
			{		
				_elements = new T[other._size]();
				_size = other._size;
				for (unsigned int i = 0; i < _size; i++)
					this->_elements[i] = other._elements[i];
			}
		}

		Array& operator=(const Array &other)
		{
			if ( this != &other)
			{
				delete [] _elements;
				if (other._size == 0)
				{
					_size = 0;
					_elements = NULL;
				}
				else
				{		
					_elements = new T[other._size]();
					_size = other._size;
					for (unsigned int i = 0; i < _size; i++)
						this->_elements[i] = other._elements[i];
				}
			}
			return *this;
		}

		~Array()
		{
			delete [] _elements;
		}

		unsigned int	size() const
		{
			return _size;
		}

		T& operator[](unsigned int index)
		{
			if (index >= _size)
				throw OutOfBoundIndex();
			return _elements[index];
		}

		const T& operator[](unsigned int index) const
		{
			if (index >= _size)
				throw OutOfBoundIndex();
			return _elements[index];
		}

		class OutOfBoundIndex : public std::exception
		{
			public :
				virtual const char* what() const throw()
				{
					return "Index out of bounds";
				}
		};

	private :
		T*				_elements;
		unsigned int	_size;

};

#endif

        numbers[i] = value;
        mirror[i] = value;

        Array<int> tmp = numbers;
